        // Konstanten für die Visualisierung

        const PADDING = 35; 

        const MIN_FREQ = 20; 

        const MAX_FREQ = 20000; 

        const DB_RANGE = 50; 

        const MIN_DB = -25;

        const MAX_DB = 25;

        

        // Array für Frequenzen zur Berechnung der Filterkurve (logarithmisch verteilt)

        const curveFrequencyHz = new Float32Array(200);



        // --- HILFSFUNKTIONEN ZUR KOORDINATENKONVERTIERUNG ---


        // Frequenz (Hz) zu X-Koordinate (Pixel)

        function getX(freq, WIDTH) {

            if (freq < MIN_FREQ) return PADDING;

            if (freq > MAX_FREQ) return WIDTH - PADDING;

            

            const logMin = Math.log10(MIN_FREQ);

            const logMax = Math.log10(MAX_FREQ);

            const logFreq = Math.log10(freq);


            return PADDING + (WIDTH - 2 * PADDING) * (logFreq - logMin) / (logMax - logMin);

        }


        // X-Koordinate (Pixel) zu Frequenz (Hz)

        function getFreq(x, WIDTH, minSliderFreq, maxSliderFreq) {

            const logMinVis = Math.log10(MIN_FREQ);

            const logMaxVis = Math.log10(MAX_FREQ);


            // Normiere X-Position (0 bis 1) innerhalb des Zeichenbereichs

            const normalizedX = (x - PADDING) / (WIDTH - 2 * PADDING);

            

            const clampedNormalizedX = Math.max(0, Math.min(1, normalizedX));


            // Berechne die logarithmische Frequenz basierend auf der visuellen Skala

            const logFreq = logMinVis + clampedNormalizedX * (logMaxVis - logMinVis);

            

            const freq = Math.pow(10, logFreq);

            

            // Beschränke auf den Slider-Bereich

            return Math.max(minSliderFreq, Math.min(maxSliderFreq, freq));

        }


        // dB-Wert zu Y-Koordinate (Pixel)

        function getY(db, HEIGHT) {

            const normalizedDb = (db - MIN_DB) / DB_RANGE; 

            return HEIGHT - PADDING - (HEIGHT - 2 * PADDING) * normalizedDb; 

        }


        // Y-Koordinate (Pixel) zu dB-Wert

        function getDb(y, HEIGHT) {

            const normalizedY = (HEIGHT - PADDING - y) / (HEIGHT - 2 * PADDING);


            const db = normalizedY * DB_RANGE + MIN_DB;

            

            // Beschränke auf den Slider-Bereich

            return Math.max(-20, Math.min(20, db));

        }

        

        // Holt die Maus- oder Touch-Position relativ zum Canvas

        function getMousePos(canvas, event) {

            const rect = canvas.getBoundingClientRect();

            // Einheitliche Behandlung für Maus- und Touch-Events

            const clientX = event.clientX || (event.touches ? event.touches[0].clientX : 0);

            const clientY = event.clientY || (event.touches ? event.touches[0].clientY : 0);

            

            const scaleX = canvas.width / rect.width;

            const scaleY = canvas.height / rect.height;

            

            return {

                x: (clientX - rect.left) * scaleX,

                y: (clientY - rect.top) * scaleY

            };

        }



        // --- 1. INITIALISIERUNG ---


        function initAudio() {

            if (!audioContext) {

                // AudioContext erstelle. Er bleibt 'suspended', bis er durch User-Interaktion resumed wird.

                audioContext = new (window.AudioContext || window.webkitAudioContext)();

                

                // Initialisiere alle Filter-Nodes

                filters.forEach(filter => {

                    filter.node = audioContext.createBiquadFilter();

                    // Setze den Typ initial, wird später durch updateFilter() überschrieben

                    filter.node.type = filter.type; 

                });


                // Analyser Node

                analyser = audioContext.createAnalyser();

                analyser.fftSize = 2048;


                // --- AUDIO ROUTING (Kette) ---

                // Low -> Mid -> High -> Analyser -> Destination

                filters[0].node.connect(filters[1].node);

                filters[1].node.connect(filters[2].node);

                filters[2].node.connect(analyser);

                analyser.connect(audioContext.destination);



                // Initialisiere das Frequenz-Array für getFrequencyResponse

                for (let i = 0; i < curveFrequencyHz.length; i++) {

                    const frac = i / curveFrequencyHz.length;

                    const logFreq = MIN_FREQ * Math.pow(MAX_FREQ / MIN_FREQ, frac);

                    curveFrequencyHz[i] = logFreq;

                }


                // Initialisiere die Filter-Parameter, damit die Kurve korrekt startet

                updateFiltersOnly();

            }

        }


        // --- 2. WEISSES RAUSCHEN GENERIEREN ---


        function createWhiteNoise() {

            const bufferSize = audioContext.sampleRate * 2; 

            const noiseBuffer = audioContext.createBuffer(1, bufferSize, audioContext.sampleRate);

            const output = noiseBuffer.getChannelData(0);


            for (let i = 0; i < bufferSize; i++) {

                output[i] = Math.random() * 2 - 1; 

            }


            noiseSource = audioContext.createBufferSource();

            noiseSource.buffer = noiseBuffer;

            noiseSource.loop = true; 

            

            // Verbinde Rauschen mit dem ersten Filter (Low)

            noiseSource.connect(filters[0].node);

        }

            // Aktualisiere AudioNode

            if (filter.node && audioContext) {

                filter.node.type = filter.type;

                filter.node.frequency.setValueAtTime(freq, audioContext.currentTime);

                filter.node.Q.setValueAtTime(q, audioContext.currentTime);

                

                // Gain nur setzen, wenn der Typ nicht Lowpass/Highpass ist

                if (!isCut) {

                    filter.node.gain.setValueAtTime(gain, audioContext.currentTime);

                } else {

                    // Setze Gain auf 0, wenn es ein Cut ist, um mögliche Fehler zu vermeiden

                    filter.node.gain.setValueAtTime(0, audioContext.currentTime);

                }

            }


            // Aktualisiere Anzeigen

            filter.gainValue.textContent = `${gain.toFixed(1)} dB`;

            filter.freqValue.textContent = `${freq.toFixed(0)} Hz`;

            filter.qValue.textContent = `${q.toFixed(1)}`;

        }

        function updateFiltersOnly() {

             filters.forEach(updateFilterUI);

             // Neuzeichnen nur, wenn das Rauschen nicht läuft, um die statische Kurve zu aktualisieren.

                 initialDraw();

             }

        // Event Listener für UI-Elemente

        filters.forEach(filter => {

            if (filter.freqSlider) filter.freqSlider.addEventListener('input', handleControlChange);

            if (filter.gainSlider) filter.gainSlider.addEventListener('input', handleControlChange);

            if (filter.qSlider) filter.qSlider.addEventListener('input', handleControlChange);

            if (filter.typeSelect) filter.typeSelect.addEventListener('change', handleControlChange);

        // --- 5. INTERAKTION (DRAG-LOGIK) ---

        

        function handleDragStart(event) {

            initAudio(); // Sicherstellen, dass die Nodes existieren


            const pos = getMousePos(canvas, event);

            

            draggedFilter = null;

            

            // Finde den Filter, dessen Anfasser getroffen wurde

            for (const filterObj of filters) {

                // Ermögliche das Ziehen für ALLE Filtertypen (auch Lowpass/Highpass)

                if (filterObj.node) {

                    const distSq = (pos.x - filterObj.x) ** 2 + (pos.y - filterObj.y) ** 2;

                    // Überprüfe, ob die Maus den Anfasser innerhalb des Radius getroffen hat

                    if (distSq < (filterObj.dragRadius * 2) ** 2) { 

                        draggedFilter = filterObj;

                        break;

                    }

                }

            }

            

            if (draggedFilter) {

                isDragging = true;

                if (event.type === 'mousedown') event.preventDefault(); 

            }

        }


        function handleDragMove(event) {

            if (!isDragging || !draggedFilter) return;

            

            const pos = getMousePos(canvas, event);

            

            if (event.type === 'touchmove') {

                event.preventDefault();

            }


            const WIDTH = canvas.offsetWidth;

            const HEIGHT = canvas.offsetHeight;

            

            // Cut-Filter können nicht über die Y-Achse gezogen werden. 

            const isDraggableOnY = draggedFilter.type === 'peaking' || draggedFilter.type === 'lowshelf' || draggedFilter.type === 'highshelf';



            // 1. Neue Frequenz (X-Achse) berechnen und Slider aktualisieren

            // DIES GILT FÜR ALLE FILTER

            const minFreq = parseFloat(draggedFilter.freqSlider.min);

            const maxFreq = parseFloat(draggedFilter.freqSlider.max);

            // Begrenze die Frequenz, um das Ziehen außerhalb des Sliders zu verhindern.

            const newFreq = getFreq(pos.x, WIDTH, minFreq, maxFreq);

            draggedFilter.freqSlider.value = newFreq.toFixed(0);

            

            // 2. Neue Gain (Y-Achse) nur für Peaking/Shelf berechnen und Slider aktualisieren

            if (isDraggableOnY) {

                const newGain = getDb(pos.y, HEIGHT);

                draggedFilter.gainSlider.value = newGain.toFixed(1);

            }

            

            // Ruft handleControlChange auf, um die UI zu aktualisieren und neu zu zeichnen

            handleControlChange(); 

        }


        function handleDragEnd() {

            isDragging = false;

            draggedFilter = null;

        }


        function attachInteractionEvents() {

            // Maus-Events

            canvas.addEventListener('mousedown', handleDragStart);

            document.addEventListener('mousemove', handleDragMove);

            document.addEventListener('mouseup', handleDragEnd);


            // Touch-Events (für mobile Geräte)

            canvas.addEventListener('touchstart', handleDragStart);

            document.addEventListener('touchmove', handleDragMove, { passive: false }); 

            document.addEventListener('touchend', handleDragEnd);

        }

        function drawAxesAndGrid(WIDTH, HEIGHT) {

            canvasCtx.font = '12px Inter';

            canvasCtx.fillStyle = '#9ca3af'; 

            canvasCtx.strokeStyle = '#374151'; 


            // Achsenlinien

            canvasCtx.beginPath();

            canvasCtx.moveTo(PADDING, PADDING);

            canvasCtx.lineTo(PADDING, HEIGHT - PADDING); 

            canvasCtx.lineTo(WIDTH - PADDING, HEIGHT - PADDING); 

            canvasCtx.lineWidth = 1;

            canvasCtx.stroke();

            canvasCtx.closePath();


            // X-Achse (Frequenzen, Logarithmisch)

            const freqLabels = [20, 50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000];


            for (let i = 0; i < freqLabels.length; i++) {

                const freq = freqLabels[i];

                const x = getX(freq, WIDTH);


                // Grid-Linie

                if (freq !== 20 && freq !== 20000) {

                    canvasCtx.beginPath();

                    canvasCtx.strokeStyle = '#1f2937';

                    canvasCtx.moveTo(x, PADDING);

                    canvasCtx.lineTo(x, HEIGHT - PADDING);

                    canvasCtx.stroke();

                }


                // Beschriftung

                const label = freq >= 1000 ? `${freq/1000}k` : `${freq}`;

                canvasCtx.textAlign = 'center';

                canvasCtx.fillText(label, x, HEIGHT - PADDING + 20);

            }


            canvasCtx.textAlign = 'center';

            canvasCtx.fillStyle = '#6b7280';

            canvasCtx.fillText('Frequenz (Hz)', WIDTH / 2, HEIGHT - 5);



            // Y-Achse (Gain, dB, Linear)

            const dbLabels = [MAX_DB, 15, 0, -15, MIN_DB];

            

            for (let i = 0; i < dbLabels.length; i++) {

                const db = dbLabels[i];

                const y = getY(db, HEIGHT);


                // Grid-Linie (Hauptlinien)

                canvasCtx.beginPath();

                canvasCtx.strokeStyle = db === 0 ? '#4b5563' : '#1f2937'; 

                canvasCtx.moveTo(PADDING, y);

                canvasCtx.lineTo(WIDTH - PADDING, y);

                canvasCtx.stroke();


                // Beschriftung

                canvasCtx.textAlign = 'right';

                canvasCtx.fillText(`${db} dB`, PADDING - 5, y + 4);

            }

            

            canvasCtx.textAlign = 'left';

            canvasCtx.fillText('Gain', 5, PADDING - 5);

        }

        

        // Berechnet den kombinierten Frequenzgang aller Filter

        function getCombinedResponse(freqs) {

            const combinedMag = new Float32Array(freqs.length).fill(1.0); // Start bei 0 dB (Magnitude 1.0)


            for (const filterObj of filters) {

                if (!filterObj.node) continue;


                const magResponse = new Float32Array(freqs.length);

                const phaseResponse = new Float32Array(freqs.length); // Phase wird nicht benötigt


                // Ruft den Frequenzgang für jeden Filter einzeln ab

                filterObj.node.getFrequencyResponse(freqs, magResponse, phaseResponse);


                // Multipliziere Magnituden für den kombinierten Effekt (serielle Kette)

                for (let i = 0; i < freqs.length; i++) {

                    combinedMag[i] *= magResponse[i];

                }

            }


            return combinedMag; 

        }


        function drawFilterCurve(WIDTH, HEIGHT) {

            if (!filters[0].node) return; // Stelle sicher, dass Nodes existieren


            const combinedMag = getCombinedResponse(curveFrequencyHz);


            // --- Kurve zeichnen ---

            canvasCtx.beginPath();

            canvasCtx.strokeStyle = '#fef08a'; // Gelb für die Kombi-Kurve

            canvasCtx.lineWidth = 3;


            for (let i = 0; i < curveFrequencyHz.length; i++) {

                const freq = curveFrequencyHz[i];

                // Magnitude in dB umrechnen: 20 * log10(Magnitude)

                const db = 20 * Math.log10(combinedMag[i]); 

                

                const x = getX(freq, WIDTH);

                const y = getY(db, HEIGHT);


                if (i === 0) {

                    canvasCtx.moveTo(x, y);

                } else {

                    canvasCtx.lineTo(x, y);

                }

            }

            canvasCtx.stroke();

            canvasCtx.closePath();

            

            // --- Anfasser (Handles) zeichnen ---

            filters.forEach(filter => {

                const freq = filter.node.frequency.value;

                const isDraggableOnY = filter.type === 'peaking' || filter.type === 'lowshelf' || filter.type === 'highshelf';

                

                let gain;

                if (isDraggableOnY) {

                    // Verwende den tatsächlichen Gain für Shelf/Peaking

                    gain = filter.node.gain.value; 

                } else {

                    // Setze den Punkt auf die 0dB Linie für Cut-Filter (nicht ziehbar)

                    gain = 0;

                }

                

                // Speichere die Position für die Drag-Logik

                filter.x = getX(freq, WIDTH);

                filter.y = getY(gain, HEIGHT);

                

                // Zeichne den Anfasser (immer, aber für Cut-Filter an 0dB)

                canvasCtx.fillStyle = filter.color; 

                canvasCtx.beginPath();

                canvasCtx.arc(filter.x, filter.y, filter.dragRadius, 0, 2 * Math.PI);

                canvasCtx.fill();

                canvasCtx.strokeStyle = '#000000';

                canvasCtx.lineWidth = 1.5;

                canvasCtx.stroke();

                canvasCtx.closePath();

                    

                // Zusätzlicher Ring, wenn der Anfasser gezogen wird

                if (draggedFilter === filter) {

                    canvasCtx.strokeStyle = '#ffffff'; 

                    canvasCtx.lineWidth = 4;

                    canvasCtx.beginPath();

                    canvasCtx.arc(filter.x, filter.y, filter.dragRadius + 4, 0, 2 * Math.PI);

                    canvasCtx.stroke();

                    canvasCtx.closePath();

                }

            });

        }

        

        function drawSpectrum(WIDTH, HEIGHT) {

            if (!analyser || !audioContext) return;


            const bufferLength = analyser.frequencyBinCount;

            const dataArray = new Uint8Array(bufferLength);

            analyser.getByteFrequencyData(dataArray);


            // NEU: Dunkles Lila mit leichter Transparenz für das Spektrum (FFT)

            canvasCtx.fillStyle = 'rgba(160, 32, 240, 0.3)'; // Medium Purple mit 30% Opazität

            canvasCtx.beginPath();

            

            canvasCtx.moveTo(PADDING, HEIGHT - PADDING);

            

            const maxVisibleIndex = Math.floor(bufferLength * (MAX_FREQ / (audioContext.sampleRate / 2)));


            for(let i = 0; i < maxVisibleIndex; i++) {

                const freq = i * audioContext.sampleRate / analyser.fftSize;

                const amplitude = dataArray[i]; 


                // Skaliere Amplitude zu einer dB-Position

                const dbSimulated = (amplitude / 255) * DB_RANGE + MIN_DB; 

                

                const x = getX(freq, WIDTH);

                const y = getY(dbSimulated, HEIGHT);

                

                canvasCtx.lineTo(x, y);

            }

            

            canvasCtx.lineTo(WIDTH - PADDING, HEIGHT - PADDING);

            canvasCtx.lineTo(PADDING, HEIGHT - PADDING);

            canvasCtx.fill();

            canvasCtx.closePath();

        }


        function drawVisualizer() {

            // Stelle sicher, dass die Canvas-Größe korrekt ist

            const WIDTH = canvas.offsetWidth;

            const HEIGHT = canvas.offsetHeight;

            canvas.width = WIDTH;

            canvas.height = HEIGHT;

            

            canvasCtx.clearRect(0, 0, WIDTH, HEIGHT);


            // 1. Achsen und Gitter zeichnen (Hintergrund)

            drawAxesAndGrid(WIDTH, HEIGHT);

            

            // Nur zeichnen, wenn das Audio läuft

            if (isPlaying && analyser) {

                // 2. Echtzeit-Spektrum zeichnen (Hintergrund-Füllung)

                drawSpectrum(WIDTH, HEIGHT);

            }


            // 3. Filterkurve und Anfasser zeichnen (Overlay)

            drawFilterCurve(WIDTH, HEIGHT);


            // Loop für kontinuierliche Aktualisierung

            if (isPlaying) {

                requestAnimationFrame(drawVisualizer);

            }

        }


        // Führe initial eine Achsenzeichnung aus und setze die Nodes auf

        function initialDraw() {

            // 1. Stelle sicher, dass die Canvas-Größe korrekt ist

            const WIDTH = canvas.offsetWidth;

            const HEIGHT = canvas.offsetHeight;

            canvas.width = WIDTH;

            canvas.height = HEIGHT;

            canvasCtx.clearRect(0, 0, WIDTH, HEIGHT);

            

            // 2. Erstelle AudioContext und Nodes, falls noch nicht geschehen,

            // damit getFrequencyResponse verwendet werden kann.

            initAudio(); 


            // 3. Achsen und Gitter zeichnen

            drawAxesAndGrid(WIDTH, HEIGHT);

            

            // 4. Führe updateFiltersOnly aus, um die UI und die Kurve für den initialen Draw vorzubereiten

            updateFiltersOnly();

            

            // 5. Zeichne die Kurve einmal mit den Standardwerten (Nodes existieren jetzt garantiert)

            drawFilterCurve(WIDTH, HEIGHT);

        }


        // Füge Interaktions-Events hinzu und initialisiere die Ansicht

        window.onload = () => {

            attachInteractionEvents();

        };


        // Behandle die Größenänderung des Fensters, um das Canvas korrekt zu rendern

        window.addEventListener('resize', () => {

             if (!isPlaying) {

                initialDraw();

            }

        });
